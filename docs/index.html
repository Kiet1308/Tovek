local currentCamera = workspace.CurrentCamera
localPlayer.CharacterAdded:Connect(function(character)
	local humanoid = character:WaitForChild("Humanoid")
	local track = humanoid:WaitForChild("Animator"):LoadAnimation(script.Sprint)
	local walkSpeed = humanoid:GetAttribute("WalkSpeed") or 16
	local clone = script.Trail:Clone()
	clone.Parent = character:WaitForChild("HumanoidRootPart")
	local v = 100

		if gameProcessed or input.KeyCode ~= Enum.KeyCode.LeftShift then
			return
		end

		clone.Enabled = true

		}):Play()

		if track.IsPlaying then

			v = math.max(v2, 0)
			return

			local v2 = v + dt * 12
			v = math.min(v2, 100)
			return
		end
	end)

		connection:Disconnect()
		connection2:Disconnect()
	end)

Inventory.luaudecompiled
local function summarize(items, p)
	local total = 0
	local values = {}

	for _, item in items do
		if item.Locked or not (not p or string.find(item.Name, p, 1, true)) then
			continue
		end

		total += item.Count

		if not (item.Rarity >= 4) then
			continue
		end

		local formatted = `{item.Name} x{item.Count}`
		table.insert(values, formatted)

		if #values >= 5 then
			break
		end
	end

	return total, table.concat(values, ", ")
end

return summarize

local Timer = {}
Timer.__index = Timer

function Timer.new(p, onComplete)
	local self = setmetatable({}, Timer)
	self.Duration = p
	self.Remaining = p
	self.Running = false
	self.OnComplete = onComplete
	return self
end

function Timer:Start()
	self.Running = true
	self.Remaining = self.Duration
end

function Timer:Tick(p)
	if not self.Running then
		return
	end

	self.Remaining -= p

	if self.Remaining <= 0 then
		self.Running = false
		self.OnComplete(self)
	end
end

return Timer

01De-inliningLuau -O2 · helper recovery

Put the helpers back.

At -O2, Luau pastes small local functions straight into their callers. The call disappears; the helper’s body turns up again and again, in places nobody wrote it.

Tovek unifies each pasted region with the helper that survived — operators, fields, upvalues and literals must all line up — and only then writes the call back. Anything it can’t prove stays exactly as compiled.

Inlined copies of award → recovered calls A third site, in the PlayerAdded handler

Real output, not a mock-up: the script was compiled with luau-compile -O2 -g1 and decompiled by the Tovek V2 v0.1 release. Recovered calls are labelled because bytecode can’t say where the original call sites were — only that these regions are exact copies of award.

02NamingEvidence, not guesswork

Names, declassified.

Shipped bytecode keeps registers, not local names. Tovek reads what the program leaves behind — the service a value came from, the event that handed it over, the method called on it, the attribute key it was read with — and names each binding for the role it plays.

When the evidence runs out, it says so: plain v and p, never a confident-sounding guess.

03Control flowLoops · guards · continue · break

Jumps in. Loops out.

Compiled Luau is a flat list of instructions and jumps. Tovek rebuilds the for loops, guards, continue and break a person would write — checking loop ownership, iterator protocol and evaluation order before it commits to a shape.

Hover an instruction to see where it landed.

  • 41 instructions
  • 8 jumps
  • → 1 for-loop · 2 continue · 1 break

If a function can’t be given source-like structure faithfully, the folder driver fails closed instead of inventing control flow. The condition not (item.Rarity >= 4) is kept as written because flipping it to < 4 is only safe when NaN is ruled out.

04CraftThe rest of the reconstruction

The details, restored.

Readable output is a thousand small decisions: colons on methods, UI trees that nest, string interpolation instead of format calls. Tovek makes them — and keeps every one of them checkable.
