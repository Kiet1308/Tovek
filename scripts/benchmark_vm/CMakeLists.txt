cmake_minimum_required(VERSION 3.19)
project(TovekBenchmarkVM LANGUAGES CXX)
if(NOT LUAU_SOURCE_DIR)
    message(FATAL_ERROR "Set LUAU_SOURCE_DIR to the pinned Luau checkout")
endif()
execute_process(COMMAND git -C "${LUAU_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE LUAU_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LUAU_REV STREQUAL "c2ec0d4e5ca50796ba174a7565298f59aa572268")
    message(FATAL_ERROR "Benchmark requires the pinned Luau revision")
endif()
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LUAU_SOURCE_DIR}" luau EXCLUDE_FROM_ALL)
add_executable(benchmark-vm main.cpp)
target_compile_features(benchmark-vm PRIVATE cxx_std_17)
target_link_libraries(benchmark-vm PRIVATE Luau.VM Luau.Common)
